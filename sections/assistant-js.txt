       // Feature calls
       async function handleAIPlan(event) {
           event.preventDefault();
           const btn = document.getElementById('btn-generate-plan');
           const result = document.getElementById('ai-result-plan');
           const content = document.getElementById('ai-content-plan');
           const loading = document.getElementById('ai-loading-plan');
           const days = document.getElementById('ai-days').value;
           
           btn.disabled = true; btn.innerText = "⏳ Đang tính...";
           result.classList.add('hidden'); loading.classList.remove('hidden');

           try {
               const data = await simulateAIResponse('plan', days);
               content.innerHTML = DOMPurify.sanitize(marked.parse(data));
               result.classList.remove('hidden');
           } catch (e) { alert("Có lỗi xảy ra. Thử lại sau!"); console.error(e); }
           finally { loading.classList.add('hidden'); btn.disabled = false; btn.innerText = "Tạo Lịch Trình"; }
       }

       async function handleAIFood(event) {
           event.preventDefault();
           const btn = document.getElementById('btn-generate-food');
           const result = document.getElementById('ai-result-food');
           const content = document.getElementById('ai-content-food');
           const loading = document.getElementById('ai-loading-food');
           const loc = document.getElementById('food-location').value || "";
           const crave = document.getElementById('food-craving').value || "";
           const inputCombined = loc + " " + crave;
           
           btn.disabled = true; btn.innerText = "⏳ Đang tìm...";
           result.classList.add('hidden'); loading.classList.remove('hidden');
           
           try {
               const data = await simulateAIResponse('food', inputCombined);
               content.innerHTML = DOMPurify.sanitize(marked.parse(data));
               result.classList.remove('hidden');
           } catch (e) { alert("Có lỗi xảy ra. Thử lại sau!"); }
           finally { loading.classList.add('hidden'); btn.disabled = false; btn.innerText = "Tìm Quán Ngon"; }
       }

       async function handleAIMechanic(event) {
           event.preventDefault();

               result.classList.remove('hidden');
           } catch (e) { alert("Có lỗi xảy ra. Thử lại sau!"); }
           finally { loading.classList.add('hidden'); btn.disabled = false; btn.innerText = "Chẩn Đoán Ngay"; }
       }

       async function handleAIProcedure(event) {
           event.preventDefault();
           const btn = document.getElementById('btn-generate-proc');
           const result = document.getElementById('ai-result-proc');
           const content = document.getElementById('ai-content-proc');
           const loading = document.getElementById('ai-loading-proc');
           const userType = document.getElementById('proc-input').value;
           if (!userType) return;

           btn.disabled = true; btn.innerHTML = "⏳ Đang tra cứu...";
           result.classList.add('hidden'); loading.classList.remove('hidden');
           
           try {
               const data = await simulateAIResponse('procedure', userType);
               content.innerHTML = DOMPurify.sanitize(marked.parse(data));
               result.classList.remove('hidden');
           } catch (e) { 
               console.error(e);
               alert("Không thể tra cứu lúc này. Vui lòng thử lại.");
           }
           finally { loading.classList.add('hidden'); btn.disabled = false; btn.innerHTML = "Tra Cứu Cọc & Lấy Xe"; }
